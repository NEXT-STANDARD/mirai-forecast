// scripts/prerender_markets.mjs:102 / scripts/generate_market_ogp.mjs:112 const worldProb = event.world_prob_yes !== undefined && event.world_prob_yes !== null ? Number(event.world_prob_yes) : 50;

undefined なので必ず 50 にフォールバックします。エラーは出ません。

同期スクリプトはすでに実際の確率を計算しています。保存していないだけです。

// scripts/sync_polymarket_cron.mjs:181-185 let probYes = 50; if (Array.isArray(parsed) && parsed[0]) probYes = Math.round(parseFloat(parsed[0]) * 100); … // :296 候補オブジェクトには入っている probYes, // しかし upsert のレコードには含まれていない
